import React, { useState } from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';

const FRAGEN = [
  { frage: 'Was ist eine Variable?', antworten: ['Ein Speicherplatz', 'Ein Bild', 'Ein Fehler'], richtig: 0 },
  { frage: 'Wofür steht HTML?', antworten: ['Home Tool', 'Hyper Text Markup Language', 'Hot Mail'], richtig: 1 },
  { frage: 'Was gibt print("Hi") aus?', antworten: ['print', 'Fehler', 'Hi'], richtig: 2 },
];

export default function App() {
  const [index, setIndex] = useState(0);
  const [streak, setStreak] = useState(0);
  const [feedback, setFeedback] = useState('');
  const frage = FRAGEN[index];

  const pruefe = (wahl) => {
    if (wahl === frage.richtig) {
      setStreak(streak + 1);
      setFeedback('Richtig!');
    } else {
      setStreak(0);
      setFeedback('Falsch');
    }
    setIndex((index + 1) % FRAGEN.length);
  };

  return (
    <View style={styles.container}>
      <Text style={styles.streak}>Streak: {streak}</Text>
      <Text style={styles.frage}>{frage.frage}</Text>
      {frage.antworten.map((text, i) => (
        <TouchableOpacity key={i} style={styles.button} onPress={() => pruefe(i)}>
          <Text style={styles.buttonText}>{text}</Text>
        </TouchableOpacity>
      ))}
      <Text style={styles.feedback}>{feedback}</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', padding: 24, backgroundColor: '#f4f4f8' },
  streak: { fontSize: 20, textAlign: 'right', marginBottom: 24 },
  frage: { fontSize: 24, fontWeight: 'bold', marginBottom: 24 },
  button: { backgroundColor: '#4f46e5', padding: 16, borderRadius: 12, marginBottom: 12 },
  buttonText: { color: 'white', fontSize: 18, textAlign: 'center' },
  feedback: { fontSize: 20, textAlign: 'center', marginTop: 16 },
});
